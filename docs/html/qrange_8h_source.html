512 static QList<QRange> cut(const QRange& range, int number)
513 {
514 QList<QRange> result;

519 if (number < 0)

558 static QRange combine(const QList<QRange>& ranges, QRangeSensitivity sensitivity = QRangeSensitivity::Sensitive)
559 {
560 auto sortedRanges = ranges;
561 QRange result;
562
563 if (sensitivity == QRangeSensitivity::Sensitive) {
564 std::sort(sortedRanges.begin(), sortedRanges.end());
565 result = sortedRanges.first();
566
567 for (const auto& range : sortedRanges) {
568 if (result == range)
569 continue;
570
571 if (result.m_direction != range.m_direction
572 && result.m_direction != QRangeDirection::Zero
573 && range.m_direction != QRangeDirection::Zero)
574 qFatal("[QRange::combine] Ranges have different directions!");
575
576 if (!result.contains(range.m_start, QRangeSensitivity::Insensitive))
577 qFatal("[QRange::combine] There is a gap between ranges!");
578
579 result.setEnd(sortedRanges[sortedRanges.count() - 1].m_end);
580 }
581 } else if (sensitivity == QRangeSensitivity::Insensitive) {
582 result = sortedRanges.first();
583
584 const auto numbers = QRange::brake(sortedRanges);
585 const auto [min, max] = std::minmax_element(numbers.begin(), numbers.end());
586 result = QRange{ *min, *max };
587
588 if (result.m_direction != QRangeDirection::Zero) {
589 int countDirectionNegative = 0;
590 int countDirectionPositive = 0;
591
592 for (const QRange& range : sortedRanges) {
593 if (range.m_direction == QRangeDirection::Minus)
594 countDirectionNegative++;
595 else if (range.m_direction == QRangeDirection::Plus)
596 countDirectionPositive++;
597 }
598
599 if (countDirectionNegative > countDirectionPositive)
600 result.setDirection(QRangeDirection::Minus);
601 else
602 result.setDirection(QRangeDirection::Plus);
603 }
604 }
605
606 return result;

845

850 else if (std::is_same_v<T, long>)

852 else if (std::is_same_v<T, long long>)
853 return value.toLongLong(&ok, base);
854 else if (std::is_same_v<T, unsigned int>)
855 return value.toUInt(&ok, base);

862 return 0;
863 }
864 } else if constexpr (std::is_floating_point_v<T>) {
